% Vigenère Cipher Decryption (Problem 9 - ocwy)
% Using correlation with English letter frequencies

% Step 1: Load ciphertexts from textbook helper file


% Step 2: Define standard English letter frequencies (a–z)
freqs = [0.08167 0.01492 0.02782 0.04253 0.12702 0.02228 0.02015 0.06094 ...
         0.06966 0.00153 0.00772 0.04025 0.02406 0.06749 0.07507 0.01929 ...
         0.00095 0.05987 0.06327 0.09056 0.02758 0.00978 0.02360 0.00150 ...
         0.01974 0.00074];

% Step 3: Set key length
m = 6;
key = zeros(1, m);  % initialize key array

% Step 4: Find best shift for each key position using correlation
for i = 1:m
    max_corr = -inf;
    best_shift = 0;

    for shift = 0:25
        vec = circshift(vigvec(ocwy, m, i), -shift);
        c = corrcoef(vec, freqs);
        if c(1,2) > max_corr
            max_corr = c(1,2);
            best_shift = shift;
        end
    end

    key(i) = -best_shift;  % negate for decryption
end

% Step 5: Decrypt the ciphertext
plaintext = vigenere(ocwy, key);

% Step 6: Display results
disp('Decrypted Plaintext Preview:');
disp(plaintext(1:min(500, length(plaintext))));
disp('Decryption Key (letters):');
disp(int2text(mod(-key, 26)));
disp('Decryption Key (numeric):');
disp(key);
